extern vec2 initial_screen_size;

vec4 position(mat4 transform_projection, vec4 vertex_position) {
    number scale_x = floor(love_ScreenSize.x / initial_screen_size.x);
    number scale_y = floor(love_ScreenSize.y / initial_screen_size.y);

    scale_x = max(1, scale_x);
    scale_y = max(1, scale_y);

    scale_x = min(scale_x, scale_y);
    scale_y = min(scale_x, scale_y);

    number offset_x = (love_ScreenSize.x - initial_screen_size.x * scale_x) / 2;
    number offset_y = (love_ScreenSize.y - initial_screen_size.y * scale_y) / 2;

    vec4 scale = vec4(scale_x, scale_y, 1, 1);
    vec4 offset = vec4(offset_x, offset_y, 0, 0);

    return transform_projection * (vertex_position * scale + offset);
}
